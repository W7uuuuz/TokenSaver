# Token-Saver 代码实现参考

## 1. 带时间戳的文本提取：Qwen3 ASR Flash

Token-Saver 依赖词级时间戳做逐秒文本密度评分。使用 Qwen3 ASR Flash 的 `enable_words=True` 参数获取毫秒级字时间戳——这也是 Round 1 校准使用的引擎。

```python
import os, time, subprocess
from pathlib import Path
import requests
from dashscope import Files
from dashscope.audio.qwen_asr import QwenTranscription

DASHSCOPE_API_KEY = os.getenv("DASHSCOPE_API_KEY")


def extract_audio(video_path: str, output_dir: str = "./audio_cache") -> str:
    """ffmpeg 提取 16kHz mono WAV"""
    os.makedirs(output_dir, exist_ok=True)
    wav_path = os.path.join(output_dir, f"{Path(video_path).stem}.wav")
    if os.path.exists(wav_path):
        return wav_path
    subprocess.run([
        "ffmpeg", "-y", "-i", video_path,
        "-vn", "-acodec", "pcm_s16le",
        "-ar", "16000", "-ac", "1",
        "-loglevel", "error", wav_path,
    ], check=True)
    return wav_path


def transcribe(video_path: str, cache_dir: str = "./asr_cache") -> dict:
    """
    Qwen3 ASR Flash 异步文件转写，返回毫秒级字时间戳。

    Returns:
        {"text": "完整转录文本",
         "words": [{"word": "苏泊尔", "start": 12.34, "end": 12.78}, ...]}
    """
    video_id = Path(video_path).stem
    cache_path = os.path.join(cache_dir, f"{video_id}_asr.json")

    # 命中缓存直接返回
    if os.path.exists(cache_path):
        import json
        with open(cache_path, "r", encoding="utf-8") as f:
            return json.load(f)

    # Step 1: 提取音频
    wav = extract_audio(video_path, os.path.join(cache_dir, "wav"))

    # Step 2: 上传到 DashScope
    upload = Files.upload(file_path=wav, purpose="transcription", api_key=DASHSCOPE_API_KEY)
    file_url = upload.output["uploaded_files"][0]["url"]

    # Step 3: 提交异步转写（enable_words=True 是关键）
    resp = QwenTranscription.async_call(
        model="qwen3-asr-flash-filetrans",
        file_url=file_url,
        api_key=DASHSCOPE_API_KEY,
        enable_words=True,
    )
    task_id = resp.output["task_id"]

    # Step 4: 轮询直到完成
    for _ in range(60):
        time.sleep(2)
        result = QwenTranscription.fetch(task_id, api_key=DASHSCOPE_API_KEY)
        if result.output["task_status"] == "SUCCEEDED":
            url = result.output["result"]["transcription_url"]
            break
    else:
        raise RuntimeError("Transcription timed out")

    # Step 5: 下载并解析
    data = requests.get(url, timeout=30).json()
    words, text_parts = [], []
    for t in data.get("transcripts", []):
        for s in t.get("sentences", []):
            if s.get("text"):
                text_parts.append(s["text"])
            for w in s.get("words", []):
                words.append({
                    "word": w["text"],
                    "start": w["begin_time"] / 1000.0,   # ms → s
                    "end":   w["end_time"]   / 1000.0,
                })

    result = {"text": "".join(text_parts), "words": words}

    # 写入缓存
    os.makedirs(cache_dir, exist_ok=True)
    import json
    with open(cache_path, "w", encoding="utf-8") as f:
        json.dump(result, f, ensure_ascii=False, indent=2)

    return result
```

## 2. 帧选择管道

ASR words → 逐秒文本构建 → 滑动窗口 → 五维评分 → 帧选择。

```python
import jieba
import cv2
import numpy as np
from typing import List

from pilot_round1.density_scorer import compute_five_dim_scores


# ---- 2.1 逐秒文本构建 ----
def build_per_second(words: List[dict], total_seconds: int):
    """词级时间戳 → 逐秒文本 + token 列表"""
    texts = [""] * total_seconds
    tokens = [[] for _ in range(total_seconds)]
    for w in words:
        sec = int(w["start"])
        if 0 <= sec < total_seconds:
            texts[sec] += w["word"]
            tokens[sec].extend(list(jieba.cut(w["word"])))
    return texts, tokens


# ---- 2.2 滑动窗口 ----
def make_windows(texts, tokens, window_sec=5, stride_sec=1):
    n = len(texts)
    win_texts, win_tokens, centers = [], [], []
    for start in range(0, n - window_sec + 1, stride_sec):
        end = start + window_sec
        win_texts.append("".join(texts[start:end]))
        merged = []
        for i in range(start, end):
            merged.extend(tokens[i])
        win_tokens.append(merged)
        centers.append(start + window_sec // 2)
    return win_texts, win_tokens, centers


# ---- 2.3 窗口评分 → 逐秒展开 ----
def expand_to_per_second(scores, centers, total_seconds):
    """最近邻插值"""
    result = [0.0] * total_seconds
    if not centers:
        return result
    for t in range(total_seconds):
        nearest = min(range(len(centers)), key=lambda i: abs(centers[i] - t))
        result[t] = scores[nearest]
    return result


# ---- 2.4 帧选择 ----
def select_frames(scores: List[float], budget: int, total_seconds: int) -> List[int]:
    """贪心 Top-K + min-gap 约束，首尾帧强制包含"""
    importance = list(scores)
    peak = max(importance)
    importance[0] = max(importance[0], peak * 1.1)
    importance[-1] = max(importance[-1], peak * 1.1)

    if budget >= total_seconds:
        return list(range(total_seconds))

    min_gap = max(1, total_seconds // (budget * 2))
    ranked = sorted(range(total_seconds), key=lambda t: importance[t], reverse=True)
    selected = set()
    for t in ranked:
        if len(selected) >= budget:
            break
        if all(abs(t - s) >= min_gap for s in selected):
            selected.add(t)
    return sorted(selected)


# ---- 2.5 视频类型预判 ----
def classify_video_type(words: List[dict], total_seconds: int) -> str:
    """停顿比 < 0.15 → 口播（回退 Uniform），否则 → 评测（走密度选择）"""
    if total_seconds <= 0:
        return "评测"
    pause_sec = 0.0
    prev_end = 0.0
    for w in words:
        gap = w["start"] - prev_end
        if gap > 1.5:
            pause_sec += gap
        prev_end = w["end"]
    return "口播" if (pause_sec / total_seconds) < 0.15 else "评测"


# ---- 2.6 完整管道 ----
def token_saver_pipeline(video_path: str, budget: int = 20) -> dict:
    # 视频时长
    cap = cv2.VideoCapture(video_path)
    fps = cap.get(cv2.CAP_PROP_FPS)
    total_seconds = int(cap.get(cv2.CAP_PROP_FRAME_COUNT) / fps)
    cap.release()

    # ASR
    asr = transcribe(video_path)
    words = asr["words"]

    # 逐秒构建 + 滑动窗口
    texts, tokens = build_per_second(words, total_seconds)
    win_texts, win_tokens, centers = make_windows(texts, tokens)

    # 五维评分
    scores = compute_five_dim_scores(win_texts, win_tokens)
    per_second = expand_to_per_second(scores["composite"], centers, total_seconds)

    # 视频类型预判 + 帧选择
    video_type = classify_video_type(words, total_seconds)
    if video_type == "口播":
        step = (total_seconds - 1) / (budget - 1) if budget > 1 else 0
        indices = sorted(set(min(int(step * i), total_seconds - 1) for i in range(budget)))
    else:
        indices = select_frames(per_second, budget, total_seconds)

    return {
        "frame_indices": indices,
        "composite_scores": per_second,
        "total_seconds": total_seconds,
        "full_text": asr["text"],
        "video_type": video_type,
    }
```

## 3. 图片提取与 MLLM 传输

Token-Saver 输出帧索引（秒）→ 从视频抽帧 → 拼合为网格图 → base64 → DashScope Qwen-VL。

```python
import base64
from io import BytesIO
import cv2
from PIL import Image, ImageDraw, ImageFont
import dashscope


# ---- 3.1 逐帧提取 ----
def extract_frames(video_path: str, frame_indices: List[int]) -> List[Image.Image]:
    """按秒索引从视频提取帧，丢失的帧返回 None"""
    cap = cv2.VideoCapture(video_path)
    fps = cap.get(cv2.CAP_PROP_FPS)
    if fps <= 0:
        cap.release(); return []

    frames = []
    for sec in sorted(frame_indices):
        cap.set(cv2.CAP_PROP_POS_FRAMES, int(sec * fps))
        ret, frame = cap.read()
        if ret:
            rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
            frames.append(Image.fromarray(rgb))
        else:
            frames.append(None)
    cap.release()
    return frames


# ---- 3.2 拼图网格 ----
def stitch_grid(frames: List[Image.Image], timestamps: List[int],
                cols: int = 5, thumb_height: int = 200) -> Image.Image:
    """
    多帧拼合成网格图，每帧下方标注时间戳。
    布局 (cols=5, 10 frames):
      [0s] [3s] [7s] [15s] [22s]
      [30s] [35s] [40s] [45s] [50s]
    """
    valid = [(ts, f) for ts, f in zip(timestamps, frames) if f is not None]
    if not valid:
        return None
    timestamps, frames = zip(*valid)
    n = len(frames)
    cols = min(cols, n)
    rows = (n + cols - 1) // cols

    # 统一缩放到 thumb_height
    thumbs = []
    for f in frames:
        ratio = thumb_height / f.height
        thumbs.append(f.resize((int(f.width * ratio), thumb_height), Image.LANCZOS))

    # 列宽取该列最宽缩略图
    col_widths = []
    for c in range(cols):
        w = max((thumbs[r * cols + c].width for r in range(rows) if r * cols + c < n), default=0)
        col_widths.append(w)

    label_h = 22
    grid_w = sum(col_widths) + (cols + 1) * 4
    grid_h = rows * (thumb_height + 4 + label_h) + 4
    grid = Image.new('RGB', (grid_w, grid_h), (25, 25, 25))
    draw = ImageDraw.Draw(grid)

    try:
        font = ImageFont.truetype("C:/Windows/Fonts/msyh.ttc", 14)
    except Exception:
        font = ImageFont.load_default()

    for c in range(cols):
        x_offset = 4 + sum(col_widths[:c]) + c * 4
        for r in range(rows):
            idx = r * cols + c
            if idx >= n:
                break
            y = 4 + r * (thumb_height + 4 + label_h)
            grid.paste(thumbs[idx], (x_offset, y))
            label = f"t={timestamps[idx]}s"
            tw = draw.textlength(label, font=font) if hasattr(draw, 'textlength') else len(label) * 8
            tx = x_offset + (col_widths[c] - tw) / 2
            draw.text((tx, y + thumb_height + 2), label, fill=(200, 200, 200), font=font)
    return grid


# ---- 3.3 发送 Qwen-VL ----
def call_qwen_vl(frames: List[Image.Image], timestamps: List[int],
                 prompt: str, model: str = "qwen3.6-flash") -> str:
    """拼图 → base64 → DashScope MultiModalConversation"""
    grid = stitch_grid(frames, timestamps)
    if grid is None:
        return "ERROR: no valid frames"

    buf = BytesIO()
    grid.save(buf, format='JPEG', quality=85)
    img_b64 = base64.b64encode(buf.getvalue()).decode()

    response = dashscope.MultiModalConversation.call(
        model=model,
        messages=[{
            "role": "user",
            "content": [
                {"image": f"data:image/jpeg;base64,{img_b64}"},
                {"text": prompt},
            ]
        }]
    )
    if response.status_code == 200:
        return response.output.choices[0].message.content[0]["text"]
    return f"ERROR: {response.code} {response.message}"
```

## 4. 端到端示例

# 一条调用完成帧选择 + MLLM 问答
result = token_saver_pipeline("video.mp4", budget=20)
print(f"视频类型: {result['video_type']}")
print(f"选中帧:   {result['frame_indices']}")

frames = extract_frames("video.mp4", result["frame_indices"])
answer = call_qwen_vl(frames, result["frame_indices"], PROMPT, model="qwen3.6-flash")
print(f"MLLM:\n{answer}")
