# 自适应帧选择：评分分布自检

> 替换原有的 `classify_video_type()` 停顿比方案

## 1. 新增函数

在 `token_saver/pipeline.py` 中新增以下三个函数：

```python
import numpy as np


def _uniform_sample(budget: int, total_seconds: int) -> list:
    """等间距均匀采样，始终包含首尾帧。"""
    if budget <= 0:
        return []
    if budget >= total_seconds:
        return list(range(total_seconds))
    if budget == 1:
        return [0]
    step = (total_seconds - 1) / (budget - 1)
    return sorted(set(
        min(int(step * i), total_seconds - 1) for i in range(budget)
    ))


def _top_k_with_gap(scores: np.ndarray, budget: int, total_seconds: int) -> list:
    """按评分降序选帧，最小间距约束防止聚类。"""
    if budget >= total_seconds:
        return list(range(total_seconds))

    scores = scores.copy()
    scores[0] = max(scores[0], scores.max() * 1.1)
    scores[-1] = max(scores[-1], scores.max() * 1.1)

    min_gap = max(1, total_seconds // (budget * 2))
    ranked = sorted(range(total_seconds), key=lambda t: scores[t], reverse=True)

    selected = set()
    for t in ranked:
        if len(selected) >= budget:
            break
        if all(abs(t - s) >= min_gap for s in selected):
            selected.add(t)

    return sorted(selected)


def select_frames_adaptive(
    composite_scores: list,
    budget: int,
    total_seconds: int,
    low_cv: float = 0.15,
    high_cv: float = 0.22,
    blend_ratio: float = 0.6,
) -> list:
    """
    根据评分分布的变异系数自动选择采样策略。

    Parameters
    ----------
    composite_scores : 逐秒重要性评分 (0-10)
    budget : 帧预算
    total_seconds : 视频秒数
    low_cv : 低于此值 → 纯均匀采样
    high_cv : 高于此值 → 纯评分选择
    blend_ratio : 灰色地带中按评分选的比例

    Returns
    -------
    list[int] : 选中帧索引（秒，0-indexed，升序）
    """
    if budget <= 0:
        return []

    scores = np.array(composite_scores, dtype=float)
    cv = float(np.std(scores) / (np.mean(scores) + 1e-10))

    if cv < low_cv:
        return _uniform_sample(budget, total_seconds)

    if cv > high_cv:
        return _top_k_with_gap(scores, budget, total_seconds)

    # 灰色地带：混合
    n_score = max(1, int(budget * blend_ratio))
    n_uniform = budget - n_score

    selected = set(_top_k_with_gap(scores, n_score, total_seconds))
    min_gap = max(1, total_seconds // (budget * 2))

    for f in _uniform_sample(budget * 2, total_seconds):
        if len(selected) >= budget:
            break
        if all(abs(f - s) >= min_gap for s in selected):
            selected.add(f)

    return sorted(selected)
```

## 2. 修改管道入口

在 `token_saver/pipeline.py` 的 `compress_video()` 函数中，将：

```python
# 旧代码：Top-K 选择
frame_indices = sorted(selected)
```

替换为：

```python
# 新代码：自适应选择
frame_indices = select_frames_adaptive(
    composite_scores, budget, total_seconds
)
```

## 4. 可调参数

| 参数 | 默认值 | 说明 |
|------|--------|------|
| `low_cv` | 0.15 | CV 低于此值 → 评分过于平坦，纯均匀采样 |
| `high_cv` | 0.22 | CV 高于此值 → 信息峰值显著，纯评分选择 |
| `blend_ratio` | 0.6 | 灰色地带中评分选择占预算的比例 |

## 5. 调用示例

```python
from token_saver import compress_video

result = compress_video("video.mp4", budget=10)
print(result["frame_indices"])
# 内部自动根据 score_cv 选择策略，无需人工指定视频类型
```
